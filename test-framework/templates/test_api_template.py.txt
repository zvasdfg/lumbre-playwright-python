import pytest
from playwright.sync_api import APIRequestContext

from automation.core.reporting import TestLogger

pytestmark = [pytest.mark.api, pytest.mark.production, pytest.mark.remote_smoke]


@pytest.mark.skip(reason="Pending exercise: review the target and replace API-000")
@pytest.mark.case("API-000", "Describe the observable public HTTP contract")
def test_descriptive_api_behavior(
    api_request_context: APIRequestContext,
    test_log: TestLogger,
) -> None:
    with test_log.step("Read a retired API route without performing a mutation"):
        response = api_request_context.get("/api/account")

    with test_log.step("Record and validate the public contract"):
        observed_value = response.status
        expected_value = 404
        test_log.values(observed_value=observed_value, expected_value=expected_value)
        assert observed_value == expected_value
